```python
import pandas as pd
from sqlalchemy import create_engine, text
from config import DB

_engine = None

def engine():
    global _engine
    if _engine is None:
        uri = (f"mysql+pymysql://{DB['user']}:{DB['password']}"
               f"@{DB['host']}:{DB['port']}/{DB['database']}?charset=utf8mb4")
        _engine = create_engine(
            uri,
            connect_args={"local_infile": True},
            pool_pre_ping=True,
        )
    return _engine

def q(sql: str, **kw) -> pd.DataFrame:
    """查询，返回 DataFrame"""
    with engine().connect() as c:
        return pd.read_sql(text(sql), c, **kw)

def run(sql: str):
    """执行 DDL / DML，自动提交"""
    with engine().begin() as c:
        c.execute(text(sql))

def run_script(path):
    """执行一个 .sql 文件（按分号切分）"""
    sql = open(path, encoding="utf-8").read()
    stmts = [s.strip() for s in sql.split(";") if s.strip()]
    with engine().begin() as c:
        for s in stmts:
            if s and not s.startswith("--"):
                c.execute(text(s))
```

```python
"""阶段 1：宽表转长表，导出 CSV 供 LOAD DATA 使用"""
import pandas as pd
from config import DATA, CSV

N_DAYS = 1941
DIM = ["id", "item_id", "dept_id", "cat_id", "store_id", "state_id"]
DCOLS = [f"d_{i}" for i in range(1, N_DAYS + 1)]

# --- 1. 读宽表（d 列用 int16，30k×1941 只占约 120MB）---
dtype = {c: "int16" for c in DCOLS}

                   usecols=DIM + DCOLS, dtype=dtype)

# --- 2. 只保留 CA_1 门店的 FOODS 品类 ---
wide = wide[(wide.store_id == "CA_1") & (wide.cat_id == "FOODS")].copy()

# --- 3. 宽转长 ---
long = wide.melt(
    id_vars=["item_id", "dept_id", "cat_id", "store_id", "state_id"],
    value_vars=DCOLS,
    var_name="d",
    value_name="units_sold",
)
long["d"] = long["d"].str.replace("d_", "", regex=False).astype("int16")
long["units_sold"] = long["units_sold"].astype("int32").astype("int16")

cal = cal.rename(columns={"date": "sale_date"})

assert long.sale_date.notna().all(), "有日期没匹配上"

# --- 5. 导出（列顺序固定，给 LOAD DATA 用）---
out_cols = ["store_id", "item_id", "dept_id", "cat_id",
            "state_id", "sale_date", "units_sold"]
```

## Step 2.1 处理价格缺失

**做什么**：新品上市前没有价格记录，先造一张稠密的价格表并填充。

**为什么要这么绕**：MySQL 8 没有 `LAST_VALUE(...) IGNORE NULLS`，不能直接用窗口函数向前填充。标准解法是「打分组号 → 组内取最大值」。

新建 `sql/02_enrich.sql`：

```sql
-- ① item × week 稠密格
DROP TABLE IF EXISTS item_week;
CREATE TABLE item_week AS
SELECT DISTINCT s.item_id, s.store_id, c.wm_yr_wk
FROM sales s JOIN calendar c ON c.sale_date = s.sale_date;

-- ② 灌价格 + 打分组号
DROP TABLE IF EXISTS price_filled;
CREATE TABLE price_filled AS
SELECT t.item_id, t.store_id, t.wm_yr_wk, p.sell_price,
       SUM(p.sell_price IS NULL) OVER (
           PARTITION BY t.item_id ORDER BY t.wm_yr_wk
       ) AS grp
FROM item_week t
LEFT JOIN sell_prices p
       ON p.item_id = t.item_id
      AND p.store_id = t.store_id
      AND p.wm_yr_wk = t.wm_yr_wk;

-- ③ 组内取最大值 = 向前填充
DROP TABLE IF EXISTS price_ff;
CREATE TABLE price_ff AS
SELECT item_id, store_id, wm_yr_wk,
       MAX(sell_price) OVER (PARTITION BY item_id, grp) AS sell_price
FROM price_filled;

-- ④ 上市前那段（组内一直为空）用首个可得价格回填
DROP TABLE IF EXISTS first_price;
CREATE TABLE first_price AS
SELECT f.item_id, p.sell_price
FROM (SELECT item_id, MIN(wm_yr_wk) AS wk
      FROM sell_prices WHERE store_id = 'CA_1'
      GROUP BY item_id) f
JOIN sell_prices p
  ON p.store_id = 'CA_1' AND p.item_id = f.item_id AND p.wm_yr_wk = f.wk;

DROP TABLE IF EXISTS price_final;
CREATE TABLE price_final AS
SELECT pf.item_id, pf.store_id, pf.wm_yr_wk,
       COALESCE(pf.sell_price, fp.sell_price) AS sell_price
FROM price_ff pf
LEFT JOIN first_price fp ON fp.item_id = pf.item_id;
```

```sql
DROP TABLE IF EXISTS abc_class;
CREATE TABLE abc_class AS
WITH rev AS (
  SELECT item_id, SUM(units_sold * sell_price) AS revenue_52w
  FROM v_sales_enriched
  WHERE sale_date > (SELECT DATE_SUB(MAX(sale_date), INTERVAL 364 DAY) FROM sales)
  GROUP BY item_id
), rk AS (
  SELECT item_id, revenue_52w,
         SUM(revenue_52w) OVER (ORDER BY revenue_52w DESC, item_id)
           / SUM(revenue_52w) OVER () AS cum_share
  FROM rev
)
SELECT item_id, revenue_52w, cum_share,
       CASE WHEN cum_share <= 0.70 THEN 'A'
            WHEN cum_share <= 0.90 THEN 'B'
            ELSE 'C' END AS abc
FROM rk;

ALTER TABLE abc_class ADD PRIMARY KEY (item_id);

DROP TABLE IF EXISTS abc_summary;
CREATE TABLE abc_summary AS
SELECT abc, COUNT(*) AS n_sku,
       ROUND(COUNT(*) / (SELECT COUNT(*) FROM abc_class) * 100, 1) AS sku_pct,
       ROUND(SUM(revenue_52w), 0) AS revenue,
       ROUND(SUM(revenue_52w) / (SELECT SUM(revenue_52w) FROM abc_class) * 100, 1) AS rev_pct
FROM abc_class GROUP BY abc ORDER BY abc;
```

# 第 5 章 特征工程（约 45 分钟，最容易出错的一章）

## Step 5.1 理解唯一铁律

> **第 t 行的任何特征，只能用第 t−1 天及以前的数据。**
> 写法上就是所有滚动窗口的右端一律 `1 PRECEDING`。

`ROWS BETWEEN 6 PRECEDING AND 1 PRECEDING` = 前 6 天，**不含当天**。
如果写成 `ROWS BETWEEN 6 PRECEDING AND CURRENT ROW`，就把当天销量算进特征了——而当天销量正是你要预测的目标，这叫**标签泄漏**，会让回测分数虚高、一上线就崩。

## Step 5.2 建特征表

新建 `sql/05_features.sql`：

```sql
DROP TABLE IF EXISTS feat_item_day;
CREATE TABLE feat_item_day AS
SELECT
  e.item_id, e.sale_date, e.dept_id,
  e.units_sold AS y,
  e.sell_price, e.wday, e.month, e.year, e.snap_CA,
  (e.event_name_1 IS NOT NULL) AS is_event,

  /* 滞后 */
  LAG(e.units_sold, 1)  OVER w AS lag_1,
  LAG(e.units_sold, 7)  OVER w AS lag_7,
  LAG(e.units_sold, 14) OVER w AS lag_14,
  LAG(e.units_sold, 28) OVER w AS lag_28,

  /* 滚动均值：右端 1 PRECEDING */
  AVG(e.units_sold) OVER (PARTITION BY e.item_id ORDER BY e.sale_date
      ROWS BETWEEN 6 PRECEDING AND 1 PRECEDING)   AS ma_7,
  AVG(e.units_sold) OVER (PARTITION BY e.item_id ORDER BY e.sale_date
      ROWS BETWEEN 27 PRECEDING AND 1 PRECEDING)  AS ma_28,
  AVG(e.units_sold) OVER (PARTITION BY e.item_id ORDER BY e.sale_date
      ROWS BETWEEN 55 PRECEDING AND 1 PRECEDING)  AS ma_56,

  /* 滚动标准差 */
  STDDEV_SAMP(e.units_sold) OVER (PARTITION BY e.item_id ORDER BY e.sale_date
      ROWS BETWEEN 6 PRECEDING AND 1 PRECEDING)   AS sd_7,
  STDDEV_SAMP(e.units_sold) OVER (PARTITION BY e.item_id ORDER BY e.sale_date
      ROWS BETWEEN 27 PRECEDING AND 1 PRECEDING)  AS sd_28,

  /* 极值 */
  MAX(e.units_sold) OVER (PARTITION BY e.item_id ORDER BY e.sale_date
      ROWS BETWEEN 27 PRECEDING AND 1 PRECEDING)  AS mx_28,
  MIN(e.units_sold) OVER (PARTITION BY e.item_id ORDER BY e.sale_date
      ROWS BETWEEN 27 PRECEDING AND 1 PRECEDING)  AS mn_28,

  /* 零销量结构（间歇性需求） */
  SUM(e.units_sold = 0) OVER (PARTITION BY e.item_id ORDER BY e.sale_date
      ROWS BETWEEN 27 PRECEDING AND 1 PRECEDING)  AS zero_days_28,

  /* 趋势 */
  AVG(e.units_sold) OVER (PARTITION BY e.item_id ORDER BY e.sale_date
      ROWS BETWEEN 6 PRECEDING AND 1 PRECEDING)
  / NULLIF(AVG(e.units_sold) OVER (PARTITION BY e.item_id ORDER BY e.sale_date
      ROWS BETWEEN 27 PRECEDING AND 1 PRECEDING), 0) AS ma7_ma28_ratio,

  /* 价格 */
  e.sell_price
  / NULLIF(AVG(e.sell_price) OVER (PARTITION BY e.item_id ORDER BY e.sale_date
      ROWS BETWEEN 27 PRECEDING AND 1 PRECEDING), 0) AS price_vs_avg28,
  e.sell_price - LAG(e.sell_price, 1) OVER w AS price_delta

FROM v_sales_enriched e
WINDOW w AS (PARTITION BY e.item_id ORDER BY e.sale_date);
```

# 第 6 章 四模型 × 10 折回测（约 2 小时）

## Step 6.1 先理解折怎么切

```
F1  训练到 2015-07-19  → 预测 2015-07-20 ~ 08-16
F2  训练到 2015-08-16  → 预测 2015-08-17 ~ 09-13
...
F10 训练到 2016-03-27  → 预测 2016-03-28 ~ 04-24
锁箱 训练到 2016-04-24 → 预测 2016-04-25 ~ 05-22（只在最后跑一次）
```

每折**重新训练**模型，不允许用未来的数据。预测 horizon 取 **42 天**，但只考核前 28 天——多出来的 14 天是给第 7 章补货仿真用的（仿真每 7 天评审一次，需要看到 t+7 ~ t+13）。

## Step 6.2 写回测脚本

新建 `src/30_backtest.py`：

```python
"""阶段 6：四模型 × 10 折 × 28 天滚动回测"""
import warnings
import numpy as np
import pandas as pd
from config import CSV, MET, PRED
from db import q

warnings.filterwarnings("ignore")

HORIZON = 42          # 预测 42 天
EVAL_H = 28           # 只考核前 28 天
N_FOLDS = 10
FIRST_ORIGIN_D = 1633  # 训练截止日

# ---------- 读数据 ----------
print("读取特征...")
feat = pd.read_parquet(CSV / "features.parquet")
cal = q("SELECT sale_date, d FROM calendar ORDER BY sale_date")
abc = q("SELECT item_id, abc FROM abc_class")

feat["sale_date"] = pd.to_datetime(feat["sale_date"])
cal["sale_date"] = pd.to_datetime(cal["sale_date"])

# 宽表：行=SKU，列=日期，值=销量
items = np.sort(feat["item_id"].unique())
dates = np.sort(feat["sale_date"].unique())
Y = (feat.pivot(index="item_id", columns="sale_date", values="y")
        .reindex(index=items, columns=dates).fillna(0).values.astype(np.float32))
d_of_date = dict(zip(cal["sale_date"], cal["d"]))
date_of_d = dict(zip(cal["d"], cal["sale_date"]))
date_idx = {d: i for i, d in enumerate(dates)}

print("SKU:", len(items), "日期:", len(dates), "矩阵:", Y.shape)

FEATURES = ["lag_1", "lag_7", "lag_14", "lag_28", "ma_7", "ma_28", "ma_56",
            "sd_7", "sd_28", "mx_28", "mn_28", "zero_days_28",
            "ma7_ma28_ratio", "price_vs_avg28", "price_delta",
            "wday", "month", "snap_CA", "is_event"]

# ---------- 四个模型 ----------

def naive_seasonal(hist, h):
    """季节朴素：去年同期（-7 天），h>7 用预测值递推"""
    out = np.zeros((hist.shape[0], h), dtype=np.float32)
    buf = np.concatenate([hist, np.zeros((hist.shape[0], h), np.float32)], axis=1)
    n = hist.shape[1]
    for k in range(h):
        v = buf[:, n + k - 7]
        buf[:, n + k] = v
        out[:, k] = v
    return out


def moving_average(hist, h, w=28):
    base = hist[:, -w:].mean(axis=1)
    return np.repeat(base[:, None], h, axis=1).astype(np.float32)


def holt_winters(hist, h):
    """Holt-Winters，失败则回落季节朴素"""
    from statsmodels.tsa.holtwinters import ExponentialSmoothing
    out = np.zeros((hist.shape[0], h), dtype=np.float32)
    fallback = 0
    for i in range(hist.shape[0]):
        s = hist[i]
        try:
            if s.sum() < 10:
                raise ValueError("too sparse")
            m = ExponentialSmoothing(s, trend="add", seasonal="add",
                                     seasonal_periods=7,
                                     initialization_method="estimated")
            fit = m.fit(optimized=True, use_boxcox=False, remove_bias=False)
            out[i] = np.maximum(fit.forecast(h), 0)
        except Exception:
            fallback += 1
            out[i] = naive_seasonal(s[None, :], h)[0]
    print(f"    Holt-Winters 回落季节朴素: {fallback} 条")
    return out


def lightgbm_recursive(origin_idx, h):
    """全局 LightGBM，递归预测"""
    import lightgbm as lgb
    # 训练集：origin 之前的所有行（特征已保证不含当天）
    train = feat[(feat["sale_date"] <= dates[origin_idx])].dropna(subset=["ma_28"])
    tr = train.dropna(subset=FEATURES)
    X, y = tr[FEATURES].values.astype(np.float32), tr["y"].values.astype(np.float32)
    model = lgb.LGBMRegressor(
        objective="tweedie", tweedie_variance_power=1.1,
        n_estimators=600, learning_rate=0.05, num_leaves=96,
        min_child_samples=50, feature_fraction=0.8,
        subsample=0.8, subsample_freq=1, n_jobs=-1, verbose=-1,
    )
    model.fit(X, y)
    # 递归：每预测一天，把预测值写回矩阵，再重算特征
    ext = Y[:, :origin_idx + 1].copy()
    preds = np.zeros((len(items), h), dtype=np.float32)
    F = feat[feat["sale_date"] == dates[origin_idx]].set_index("item_id")[FEATURES]
    cur = F.reindex(items).values.astype(np.float32)
    for k in range(h):
        p = np.maximum(model.predict(cur), 0)
        preds[:, k] = p
        ext = np.concatenate([ext, p[:, None]], axis=1)
        n = ext.shape[1]
        cur = np.column_stack([
            np.roll(p, 1), np.roll(p, 7), np.roll(p, 14), np.roll(p, 28),
            ext[:, -7:].mean(1), ext[:, -28:].mean(1), ext[:, -56:].mean(1),
            ext[:, -7:].std(1), ext[:, -28:].std(1),
            ext[:, -28:].max(1), ext[:, -28:].min(1),
            (ext[:, -28:] == 0).sum(1),
            ext[:, -7:].mean(1) / np.maximum(ext[:, -28:].mean(1), 1e-6),
            cur[:, 13], cur[:, 14], cur[:, 15], cur[:, 16], cur[:, 17], cur[:, 18],
        ]).astype(np.float32)
        cur = np.nan_to_num(cur, nan=0.0, posinf=0.0, neginf=0.0)
    return preds


# ---------- 主循环 ----------
results, records = [], []
for k in range(N_FOLDS):
    origin_d = FIRST_ORIGIN_D + 28 * k
    start_d, end_d = origin_d + 1, origin_d + EVAL_H
    o_idx = date_idx[date_of_d[origin_d]]
    hist = Y[:, :o_idx + 1]
    print(f"\n=== F{k+1} 训练至 {date_of_d[origin_d].date()} 预测 {date_of_d[start_d].date()}~{date_of_d[end_d].date()}")

    preds = {
        "naive":  naive_seasonal(hist, HORIZON),
        "ma28":   moving_average(hist, HORIZON),
        "hw":     holt_winters(hist, HORIZON),
        "lgbm":   lightgbm_recursive(o_idx, HORIZON),
    }
    s_idx = date_idx[date_of_d[start_d]]
    truth = Y[:, s_idx: s_idx + EVAL_H]

    for name, P in preds.items():
        ph = P[:, :EVAL_H]
        err = np.abs(truth - ph)
        wmape = err.sum() / max(truth.sum(), 1)
        bias = (ph - truth).sum() / max(truth.sum(), 1)
        mae = err.mean()
        rmse = np.sqrt(((truth - ph) ** 2).mean())
        records.append(dict(fold=k+1, model=name, wmape=wmape, bias=bias,
                            mae=mae, rmse=rmse))
        print(f"  {name:6s} WMAPE={wmape:.4f} Bias={bias:+.4f}")
        if name == "lgbm":
            ok = o_idx + 1 + EVAL_H
            df = pd.DataFrame(P[:, :EVAL_H], index=items,
                              columns=[dates[i] for i in range(s_idx, s_idx + EVAL_H)])
            results.append(df.stack().rename("yhat").reset_index()
                           .assign(model=name, fold=k+1))

met = pd.DataFrame(records)
met.to_csv(MET / "metrics_fold.csv", index=False, encoding="utf-8-sig")

# 分 ABC
abc_map = dict(zip(abc["item_id"], abc["abc"]))
g = met.groupby("model")["wmape"].mean().sort_values()
print("\n=== 各模型 10 折平均 WMAPE ===")
print(g.to_string())
g.to_frame("wmape_mean").to_csv(MET / "model_compare.csv", encoding="utf-8-sig")
```

# 第 7 章 (R,S) 补货仿真（约 45 分钟）

## Step 7.1 先想清楚三个公式

```
保护期 = R + LT = 7 + 7 = 14 天
订货点式补货：每 7 天评审一次，订到 S
S = 保护期需求预测 + 安全库存 SS

安全库存怎么定？解这个方程（不是取分位数！）：
    β(SS) = 1 − E[(e − SS)⁺] / E[D₁₄] = 0.95
其中 e = 实际需求 − 预测，来自回测残差
```

**为什么不能取 95 分位**：取分位数的含义是「95% 的概率需求不超过 S」，那是**不缺货概率**（cycle service level），不是 fill rate。要控 fill rate 必须解上面这个期望缺货方程。

## Step 7.2 写仿真脚本

新建 `src/40_simulate.py`：

```python
"""阶段 7：(R,S) 补货仿真，固定 SKU 级 fill rate = 95%"""
import numpy as np
import pandas as pd
from config import CSV, SIM, MET
from db import q

R, LT = 7, 7
PROT = R + LT          # 14
TARGET_BETA = 0.95
N_DAYS = 280

feat = pd.read_parquet(CSV / "features.parquet")
items = np.sort(feat["item_id"].unique())
n_sku = len(items)

# ---- 用第 6 章回测得到的预测；这里以 ma28 与 naive 为例 ----
# 实际使用时把 pred_backtest 从库里读出来
price = q("""SELECT item_id, AVG(sell_price) p FROM v_sales_enriched
             GROUP BY item_id""").set_index("item_id")["p"].reindex(items).values
abc = q("SELECT item_id, abc FROM abc_class").set_index("item_id")["abc"].reindex(items).values


def solve_ss(errors_14, demand_14):
    """二分法解 beta(SS)=TARGET_BETA；errors_14 形状 (n_sku, n_win)"""
    ss = np.zeros(n_sku)
    for i in range(n_sku):
        e = errors_14[i][~np.isnan(errors_14[i])]
        d = max(demand_14[i], 1e-6)
        if len(e) == 0:
            continue
        lo, hi = 0.0, max(e.max(), 0.0) + 1.0
        for _ in range(40):
            mid = (lo + hi) / 2
            beta = 1 - np.maximum(e - mid, 0).mean() / d
            if beta < TARGET_BETA:
                lo = mid
            else:
                hi = mid
        ss[i] = hi
    return ss


def simulate(demand, forecast, ss, target_orders=None):
    """
    demand:   (n_sku, T) 实际需求（这里是历史销量，作为需求代理）
    forecast: (n_sku, T) 事先给出的 14 天前瞻预测
    ss:       (n_sku,)  安全库存
    返回逐指标字典
    """
    T = demand.shape[1]
    I = np.zeros(n_sku)                 # 在库
    pipe = np.zeros((n_sku, LT))        # 在途
    on_hand = np.zeros((n_sku, T))
    shipped = np.zeros((n_sku, T))
    unmet = np.zeros((n_sku, T))
    orders = np.zeros((n_sku, T))

    for t in range(T):
        I += pipe[:, t % LT]                      # 1 到货
        ship = np.minimum(I, demand[:, t])        # 2 满足需求（丢单）
        I -= ship
        shipped[:, t] = ship
        unmet[:, t] = demand[:, t] - ship
        on_hand[:, t] = I

        if t % R == 0:                            # 3 评审订货
            IP = I + pipe.sum(axis=1)
            S = forecast[:, t] + ss
            qty = np.maximum(0, S - IP)
            pipe[:, t % LT] += qty
            orders[:, t] = qty

    warm = 28
    beta = shipped[:, warm:].sum() / max(demand[:, warm:].sum(), 1e-6)
    beta_sku = shipped[:, warm:].sum(1) / np.maximum(demand[:, warm:].sum(1), 1e-6)
    avg_inv = on_hand[:, warm:].mean(1)
    dos = avg_inv / np.maximum(demand[:, warm:].mean(1), 1e-6)      # 需求口径
    dos_sales = avg_inv / np.maximum(shipped[:, warm:].mean(1), 1e-6)  # 已售口径（对照）

    GM, HOLD_R = 0.25, 0.22
    unit_cost = price * (1 - GM)
    h_day = unit_cost * HOLD_R / 365
    hold_cost = (on_hand[:, warm:] * h_day[:, None]).sum()
    loss_cost = (unmet[:, warm:] * (price * GM)[:, None]).sum()

    return dict(
        sku_beta_median=float(np.median(beta_sku)),
        sys_beta=float(beta),
        avg_inv=float(avg_inv.sum()),
        avg_inv_value=float((avg_inv * price).sum()),
        dos_days=float(np.average(dos, weights=np.maximum(demand[:, warm:].sum(1), 1e-6))),
        dos_sales_days=float(np.average(dos_sales, weights=np.maximum(demand[:, warm:].sum(1), 1e-6))),
        stockout_days=int((unmet[:, warm:] > 0).sum()),
        n_orders=int((orders[:, warm:] > 0).sum()),
        hold_cost=float(hold_cost),
        loss_cost=float(loss_cost),
        total_cost=float(hold_cost + loss_cost),
    )
```
